cmake_minimum_required(VERSION 3.25)
project(cpp_mlx_feasibility LANGUAGES CXX)
find_package(MLX CONFIG REQUIRED)
# The installed wheel exports its builder's absolute SDK framework paths.
# Resolve those same frameworks against this machine's SDK; no runtime/backend change.
if(APPLE)
  get_target_property(probe_mlx_links mlx INTERFACE_LINK_LIBRARIES)
  set(probe_local_links "")
  foreach(probe_link IN LISTS probe_mlx_links)
    if(probe_link MATCHES "^/.*/([^/]+)\\.framework$" AND NOT EXISTS "${probe_link}")
      set(probe_framework_name "${CMAKE_MATCH_1}")
      find_library(PROBE_FRAMEWORK_${probe_framework_name} NAMES "${probe_framework_name}" REQUIRED)
      message(STATUS "Resolve MLX framework ${probe_framework_name} using local SDK")
      list(APPEND probe_local_links "${PROBE_FRAMEWORK_${probe_framework_name}}")
    else()
      list(APPEND probe_local_links "${probe_link}")
    endif()
  endforeach()
  set_target_properties(mlx PROPERTIES INTERFACE_LINK_LIBRARIES "${probe_local_links}")
endif()
add_executable(cpp_probe probe.cpp)
target_compile_features(cpp_probe PRIVATE cxx_std_17)
target_compile_options(cpp_probe PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(cpp_probe PRIVATE mlx)
